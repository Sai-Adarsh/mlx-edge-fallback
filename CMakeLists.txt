cmake_minimum_required(VERSION 3.27)
project(mlx_edge_fallback LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EDGE_ENABLE_MLX "Build the Apple Silicon local model backend" OFF)
set(TOKENIZERS_CPP_SOURCE "" CACHE PATH "Path to a recursive tokenizers-cpp checkout")

find_package(CURL REQUIRED)
find_package(nlohmann_json 3.12 CONFIG QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  FetchContent_Declare(nlohmann_json
    GIT_REPOSITORY https://github.com/nlohmann/json.git
    GIT_TAG v3.12.0
    GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(nlohmann_json)
endif()

add_executable(edge_fallback src/main.cpp)
target_link_libraries(edge_fallback PRIVATE CURL::libcurl nlohmann_json::nlohmann_json)

if(EDGE_ENABLE_MLX)
  if(NOT TOKENIZERS_CPP_SOURCE)
    message(FATAL_ERROR "Set TOKENIZERS_CPP_SOURCE to a recursive tokenizers-cpp checkout.")
  endif()
  find_package(MLX CONFIG REQUIRED)
  block(SCOPE_FOR VARIABLES)
    # The tokenizer checkout's msgpack submodule declares an older CMake minimum.
    if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0 AND NOT DEFINED CMAKE_POLICY_VERSION_MINIMUM)
      set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
    endif()
    add_subdirectory("${TOKENIZERS_CPP_SOURCE}" tokenizers-cpp EXCLUDE_FROM_ALL)
  endblock()
  target_sources(edge_fallback PRIVATE src/mlx_model.cpp)
  target_compile_definitions(edge_fallback PRIVATE EDGE_HAS_MLX=1)
  target_link_libraries(edge_fallback PRIVATE mlx tokenizers_cpp)
endif()

include(CTest)
option(EDGE_TEST_LONG_DECODE "Run long prompt and decode parity against EDGE_TEST_MODEL" OFF)
set(EDGE_TEST_MODEL "" CACHE PATH "Optional mlx-community/Llama-3.2-3B-Instruct-4bit checkpoint for CTest")
if(EDGE_TEST_LONG_DECODE AND NOT (BUILD_TESTING AND EDGE_ENABLE_MLX AND EDGE_TEST_MODEL))
  message(FATAL_ERROR "EDGE_TEST_LONG_DECODE requires BUILD_TESTING, EDGE_ENABLE_MLX, and EDGE_TEST_MODEL.")
endif()

if(BUILD_TESTING AND UNIX)
  add_executable(edge_cli_test tests/edge_cli.cpp)
  target_link_libraries(edge_cli_test PRIVATE nlohmann_json::nlohmann_json)
  if(EDGE_ENABLE_MLX)
    target_sources(edge_cli_test PRIVATE tests/mlx_fixture.cpp)
    target_compile_definitions(edge_cli_test PRIVATE EDGE_TEST_MLX=1)
    target_link_libraries(edge_cli_test PRIVATE mlx)
    add_test(NAME edge_dense_checkpoint COMMAND edge_cli_test $<TARGET_FILE:edge_fallback> dense)
    set_tests_properties(edge_dense_checkpoint PROPERTIES LABELS metal TIMEOUT 90)
    set(args_mode mlx-arguments)
  else()
    set(args_mode arguments)
  endif()
  add_test(NAME edge_arguments COMMAND edge_cli_test $<TARGET_FILE:edge_fallback> ${args_mode})
  add_test(NAME edge_cloud COMMAND edge_cli_test $<TARGET_FILE:edge_fallback> cloud)
  set_tests_properties(edge_arguments edge_cloud PROPERTIES TIMEOUT 30)
  if(EDGE_ENABLE_MLX AND EDGE_TEST_MODEL)
    add_test(NAME edge_mlx COMMAND edge_cli_test $<TARGET_FILE:edge_fallback> mlx "${EDGE_TEST_MODEL}")
    add_test(NAME edge_mlx_cooldown COMMAND edge_cli_test $<TARGET_FILE:edge_fallback> mlx-cooldown "${EDGE_TEST_MODEL}")
    add_test(NAME edge_mlx_prefix COMMAND edge_cli_test $<TARGET_FILE:edge_fallback> mlx-prefix "${EDGE_TEST_MODEL}")
    set_tests_properties(edge_mlx edge_mlx_cooldown edge_mlx_prefix PROPERTIES LABELS metal TIMEOUT 120)
    if(EDGE_TEST_LONG_DECODE)
      find_program(EDGE_JQ jq REQUIRED)
      add_test(NAME edge_mlx_long_decode
        COMMAND "${CMAKE_COMMAND}" -E env "EDGE_JQ=${EDGE_JQ}"
                sh "${CMAKE_CURRENT_SOURCE_DIR}/bench/long_decode_ab.sh"
                $<TARGET_FILE:edge_fallback> "${EDGE_TEST_MODEL}" 1024 1)
      set_tests_properties(edge_mlx_long_decode PROPERTIES LABELS "metal;long" TIMEOUT 180)
    endif()
  endif()
endif()
